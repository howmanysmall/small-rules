---
title: "No Native Properties Spread"
---

import RulePage from "$components/rule-page.astro";

<RulePage rule="no-native-properties-spread">
	<Fragment slot="rationale">
		<p>
			Spreading a static bag into `nativeProperties`, or onto a JSX tag through a temp object, builds a new table
			on each render. Waste if the bag never changes.
		</p>
		<p>
			This catches a spread from an object literal or a module-level constant, the kind that won't change. Put
			the fields on the tag and skip the copy.
		</p>
	</Fragment>
</RulePage>
