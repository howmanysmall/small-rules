title: "No Reflect Apply"
---

import RulePage from "$components/rule-page.astro";

<RulePage rule="no-reflect-apply">
	<Fragment slot="rationale">
		<p>
			Ported from <a href="https://github.com/dmmulroy/anti-slop">dmmulroy/anti-slop</a>. `Reflect.apply` bypasses
			the function type that would normally document and check the call. Call the typed function directly. When
			the callee or the argument list is only known at runtime, put that call behind a named interface.
		</p>
	</Fragment>
</RulePage>
