import RulePage from "$components/rule-page.astro";

<RulePage rule="require-safety-comment-for-type-assertion">
	<Fragment slot="rationale">
		<p>
			Ported from <a href="https://github.com/dmmulroy/anti-slop">dmmulroy/anti-slop</a>. A cast is an unchecked
			claim. A nearby `SAFETY:` comment records the invariant that makes that claim legitimate; `as const` is
			exempt because it doesn't claim an unrelated runtime shape.
		</p>
		<p>
			A comment that turns off `typescript/no-unsafe-type-assertion` also counts as justification, so you don't
			write the same sentence twice. That only works when the comment actually covers the diagnostic.{" "}
			`oxlint-tsgolint@7.0.2002` narrowed the report onto the `as Type` syntax, so a `disable-next-line` on the
			line preceding a multi-line object does nothing while the `as` still fails. Tell the rule which world you're
			in with{" "}
			<a href="/small-rules/configuration/#plugin-settings">`{'settings["small-rules"].tsgolintVersion'}`</a>. If
			you omit it, the lookup checks `node_modules/oxlint-tsgolint` and `pnpm-workspace.yaml`. It also checks
			`package.json`.
		</p>
		<p>
			A described disable of `typescript/no-unsafe-enum-assignment` counts too, since that rule also flags a cast
			to an enum type. To count other rules, list their IDs in the `suppressionRules` option. The option replaces
			the default list (`typescript/no-unsafe-type-assertion` and `typescript/no-unsafe-enum-assignment`), so
			repeat the ones you still want.
		</p>
	</Fragment>
</RulePage>
