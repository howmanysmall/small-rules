---
title: "No Module Mocking"
---

import RulePage from "$components/rule-page.astro";

<RulePage rule="no-module-mocking">
	<Fragment slot="rationale">
		<p>
			Ported from <a href="https://github.com/dmmulroy/anti-slop">dmmulroy/anti-slop</a>. Module mocks make tests
			depend on import machinery instead of a parameter the production code already takes. Use a real
			interface, a service layer, or an in-memory fake that behaves like the module.
		</p>
	</Fragment>
</RulePage>
